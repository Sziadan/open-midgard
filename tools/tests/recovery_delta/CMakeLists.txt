set(recovery_source "${CMAKE_CURRENT_LIST_DIR}/../../../src/gamemode/GameModePacket.cpp")
file(READ "${recovery_source}" recovery_code)
function(recovery_extract out begin end)
    string(FIND "${recovery_code}" "${begin}" start)
    if(start LESS 0)
        message(FATAL_ERROR "Cannot locate recovery source function: ${begin}")
    endif()
    string(SUBSTRING "${recovery_code}" ${start} -1 tail)
    string(FIND "${tail}" "${end}" length)
    if(length LESS 0)
        message(FATAL_ERROR "Cannot locate recovery source boundary: ${end}")
    endif()
    string(SUBSTRING "${tail}" 0 ${length} body)
    string(REGEX MATCHALL "\n[A-Za-z_][A-Za-z0-9_:<>*& ,\t]*[ \t]+[A-Za-z_][A-Za-z0-9_]*(::[A-Za-z_][A-Za-z0-9_]*)*[ \t]*\\([^;{}]*\\)[ \t\r\n]*\\{" definitions "\n${body}")
    list(LENGTH definitions count)
    if(NOT count EQUAL 1)
        message(FATAL_ERROR "Recovery source boundary encloses ${count} functions: ${begin}")
    endif()
    set(${out} "${body}" PARENT_SCOPE)
endfunction()
recovery_extract(effect "int ResolveRecoveryEffectId(" "u32 ResolveRecoveryNumberColor(")
recovery_extract(color "u32 ResolveRecoveryNumberColor(" "void EmitRecoveryNumber(")
recovery_extract(number "void EmitRecoveryNumber(" "std::string ToLowerAsciiStatus(")
recovery_extract(local_actor "bool IsLocalPlayerActor(const CGameMode& mode, u32 gid)" "CGameActor* FindRuntimeActorForVanish(")
recovery_extract(find_actor "CGameActor* FindRuntimeActorForVanish(CGameMode& mode, u32 gid)" "void BeginActorDeath(")
recovery_extract(recovery "void HandleRecovery(CGameMode& mode, const PacketView& packet)" "void HandleAttackRange(")
recovery_extract(resurrection "void HandleActorResurrection(CGameMode& mode, const PacketView& packet)" "void LogActorPacketSample(")

    "${effect}\n${color}\n${number}\n${local_actor}\n${find_actor}\n${recovery}\n${resurrection}")
